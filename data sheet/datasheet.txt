    // Utility: create a list item DOM node
    function createItem(sheet) {
      const a = document.createElement('a');
      a.href = sheet.filename || '#';
      a.className = 'group block hover:bg-gray-50 focus:outline-none focus:ring-2 focus:ring-blue-500';
      a.setAttribute('download', '');
      a.setAttribute('aria-label', `${sheet.title} - Download`);

      a.innerHTML = `
        <div class="p-4 grid grid-cols-12 gap-4 items-center">
          <div class="col-span-1 flex items-center justify-center">
            <svg class="doc-icon" viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">

              <path d="M14 2v6h6" stroke="currentColor" stroke-width="1.2" stroke-linecap="round" stroke-linejoin="round"></path>
            </svg>
          </div>

          <div class="col-span-7">
            <div class="font-medium text-gray-900">${sheet.title}</div>
            <div class="text-sm text-gray-500">${sheet.id} • ${sheet.application}</div>
          </div>

          <div class="col-span-3 text-sm text-gray-600">
            <div>${sheet.notes || ''}</div>
            <div class="mt-1 text-xs text-gray-400">Category: ${sheet.category}</div>
          </div>

          <div class="col-span-1 text-right">

            </svg>
          </div>
        </div>
      `;

      return a;
    }

    function render(list) {
      const container = document.getElementById('list');
      container.innerHTML = '';
      if (list.length === 0) {
        container.innerHTML = '<div class="p-8 text-center text-gray-500">No data sheets found.</div>';
      } else {
        list.forEach(item => {
          const node = createItem(item);
          container.appendChild(node);
          const hr = document.createElement('div');
          hr.className = 'border-t border-gray-100';
          container.appendChild(hr);
        });
      }
      document.getElementById('count').textContent = list.length;
    }

    // Initial render
    render(dataSheets);

    // Search & filter
    const searchInput = document.getElementById('search');
    const categorySelect = document.getElementById('category');

    function applyFilters() {
      const q = searchInput.value.trim().toLowerCase();
      const cat = categorySelect.value;
      const filtered = dataSheets.filter(s => {
        const matchesCat = (cat === 'all') || s.category === cat;
        const matchesQ = q === '' || [s.title, s.id, s.application, s.notes].join(' ').toLowerCase().includes(q);
        return matchesCat && matchesQ;
      });
      render(filtered);
    }

    searchInput.addEventListener('input', applyFilters);
    categorySelect.addEventListener('change', applyFilters);

    // Set dynamic year, and optionally hook download-all
    document.getElementById('year').textContent = new Date().getFullYear();

    // Download all: creates a simple zip-style behavior by opening all links in new tabs (replace with real zip on server)
    document.getElementById('download-all').addEventListener('click', (e) => {
      e.preventDefault();
      // If you have a server-side ZIP of all datasheets, set the href to that file instead.
      dataSheets.forEach(s => {
        if (s.filename) window.open(s.filename, '_blank');
      });
    });
